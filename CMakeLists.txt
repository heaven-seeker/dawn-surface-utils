cmake_minimum_required(VERSION 3.21)
project(dawn_utils VERSION 1.0.0 LANGUAGES CXX)

if(EMSCRIPTEN)
    message(FATAL_ERROR "dawn_utils supports native Dawn window surfaces")
endif()

option(DAWN_UTILS_WITH_GLFW "Build GLFW surface utilities" ON)
option(DAWN_UTILS_WITH_SDL "Build SDL3 surface utilities" ON)
option(DAWN_UTILS_BUILD_EXAMPLES "Build minimal window examples" OFF)
option(DAWN_UTILS_BUILD_TESTS "Build tests requiring a display and GPU" OFF)
set(DAWN_UTILS_DAWN_TARGET "" CACHE STRING "Dawn C API target; auto-detected when empty")

find_package(Threads REQUIRED)
if(NOT DAWN_UTILS_DAWN_TARGET)
    foreach(name IN ITEMS dawn::webgpu_dawn webgpu_dawn webgpu)
        if(TARGET ${name})
            set(DAWN_UTILS_DAWN_TARGET ${name})
            break()
        endif()
    endforeach()
    if(NOT DAWN_UTILS_DAWN_TARGET)
        find_package(Dawn CONFIG REQUIRED)
        set(DAWN_UTILS_DAWN_TARGET dawn::webgpu_dawn)
    endif()
endif()
if(NOT TARGET ${DAWN_UTILS_DAWN_TARGET})
    message(FATAL_ERROR "DAWN_UTILS_DAWN_TARGET must name an existing target")
endif()

add_library(dawn_utils_core INTERFACE)
add_library(dawn_utils::core ALIAS dawn_utils_core)
target_compile_features(dawn_utils_core INTERFACE cxx_std_20)
target_include_directories(dawn_utils_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(dawn_utils_core INTERFACE ${DAWN_UTILS_DAWN_TARGET})

if(DAWN_UTILS_WITH_GLFW)
    if(NOT TARGET glfw AND NOT TARGET glfw3::glfw)
        find_package(glfw3 3.4 CONFIG REQUIRED)
    endif()
    if(TARGET glfw)
        set(dawn_utils_glfw_dependency glfw)
    else()
        set(dawn_utils_glfw_dependency glfw3::glfw)
    endif()
    add_library(dawn_utils_glfw STATIC src/glfw_surface.cpp)
    add_library(dawn_utils::glfw ALIAS dawn_utils_glfw)
    target_link_libraries(dawn_utils_glfw PUBLIC dawn_utils::core ${dawn_utils_glfw_dependency})
    set_target_properties(dawn_utils_glfw PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)

    if(WIN32)
        target_link_libraries(dawn_utils_glfw PRIVATE user32)
    elseif(APPLE)
        enable_language(OBJCXX)
        target_sources(dawn_utils_glfw PRIVATE src/glfw_metal.mm)
        set_target_properties(dawn_utils_glfw PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
        target_compile_options(dawn_utils_glfw PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
        target_link_libraries(dawn_utils_glfw PRIVATE "-framework Cocoa" "-framework QuartzCore")
    elseif(UNIX)
        set(dawn_utils_x11_default ON)
        set(dawn_utils_wayland_default ON)
        if(DEFINED GLFW_BUILD_X11)
            set(dawn_utils_x11_default ${GLFW_BUILD_X11})
        endif()
        if(DEFINED GLFW_BUILD_WAYLAND)
            set(dawn_utils_wayland_default ${GLFW_BUILD_WAYLAND})
        endif()
        option(DAWN_UTILS_GLFW_X11 "Use GLFW's X11 native API" ${dawn_utils_x11_default})
        option(DAWN_UTILS_GLFW_WAYLAND "Use GLFW's Wayland native API" ${dawn_utils_wayland_default})
        if(NOT DAWN_UTILS_GLFW_X11 AND NOT DAWN_UTILS_GLFW_WAYLAND)
            message(FATAL_ERROR "Enable an X11 or Wayland GLFW backend")
        endif()
        target_compile_definitions(dawn_utils_glfw PRIVATE
            DAWN_UTILS_GLFW_X11=$<BOOL:${DAWN_UTILS_GLFW_X11}>
            DAWN_UTILS_GLFW_WAYLAND=$<BOOL:${DAWN_UTILS_GLFW_WAYLAND}>)
        if(DAWN_UTILS_GLFW_X11)
            find_package(X11 REQUIRED)
            find_path(dawn_utils_xrandr_headers X11/extensions/Xrandr.h REQUIRED)
            target_include_directories(dawn_utils_glfw PRIVATE ${X11_INCLUDE_DIR} ${dawn_utils_xrandr_headers})
        endif()
        if(DAWN_UTILS_GLFW_WAYLAND)
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(DAWN_UTILS_WL REQUIRED wayland-client)
            target_include_directories(dawn_utils_glfw PRIVATE ${DAWN_UTILS_WL_INCLUDE_DIRS})
            target_compile_options(dawn_utils_glfw PRIVATE ${DAWN_UTILS_WL_CFLAGS_OTHER})
        endif()
    else()
        message(FATAL_ERROR "Unsupported GLFW platform")
    endif()
endif()

if(DAWN_UTILS_WITH_SDL)
    if(NOT TARGET SDL3::SDL3)
        find_package(SDL3 3.2 CONFIG REQUIRED)
    endif()
    add_library(dawn_utils_sdl STATIC src/sdl_surface.cpp)
    add_library(dawn_utils::sdl ALIAS dawn_utils_sdl)
    target_link_libraries(dawn_utils_sdl PUBLIC dawn_utils::core SDL3::SDL3)
    set_target_properties(dawn_utils_sdl PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
endif()

function(dawn_utils_deploy_runtime target)
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND}
                "-DRUNTIME_FILES=$<TARGET_RUNTIME_DLLS:${target}>"
                "-DOUTPUT_DIR=$<TARGET_FILE_DIR:${target}>"
                -P "${PROJECT_SOURCE_DIR}/cmake/DeployRuntime.cmake"
            VERBATIM)
    endif()
endfunction()

if(DAWN_UTILS_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(DAWN_UTILS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
